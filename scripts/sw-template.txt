/*
 * Offline-App-Shell für isiHunt.
 *
 * Navigationen sind network-first, damit ein neuer Deploy sofort sichtbar
 * wird. Fällt das Netz aus, kommt die zuletzt geladene index.html aus dem
 * Cache. Die Build-Dateien werden bereits bei der Installation vorgehalten;
 * Supabase- und sonstige Fremdanfragen bleiben außen vor.
 */

const CACHE_NAME = __ISIHUNT_CACHE_NAME__;
const PRECACHE_URLS = __ISIHUNT_PRECACHE_URLS__;

/*
 * Das Versionsmanifest beantwortet die Frage "gibt es einen neuen Deploy?".
 * Aus dem Cache beantwortet es immer die eigene, alte Version - die Antwort
 * waere damit strukturell nutzlos. Es bleibt deshalb komplett aus Precache
 * und Cache-first-Zweig heraus (AUDIT_2026-09-05, Befund 7).
 */
const VERSION_MANIFEST_PATH = new URL('./version.json', self.location).pathname;

function cachedResponse(request) {
  const url = new URL(request.url);
  return caches.match(`${url.origin}${url.pathname}`, { ignoreSearch: true });
}

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches
      .open(CACHE_NAME)
      .then((cache) => cache.addAll(PRECACHE_URLS))
      .then(() => self.skipWaiting()),
  );
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches
      .keys()
      .then((keys) =>
        Promise.all(
          keys
            .filter((key) => key.startsWith('isihunt-app-shell-') && key !== CACHE_NAME)
            .map((key) => caches.delete(key)),
        ),
      )
      .then(() => self.clients.claim()),
  );
});

self.addEventListener('fetch', (event) => {
  const request = event.request;
  if (request.method !== 'GET') return;

  const url = new URL(request.url);
  if (url.origin !== self.location.origin) return;

  if (request.mode === 'navigate') {
    event.respondWith(
      fetch(request, { cache: 'no-store' })
        .then((response) => {
          // Nur eine erfolgreiche Antwort darf die App-Shell ersetzen. Ein
          // HTTP 503/404 ist fuer `fetch` kein Fehler, landete hier also
          // frueher im Cache - der naechste Offline-Start zeigte dann die
          // Fehlerseite statt der App (AUDIT_2026-09-05, Befund 10).
          if (!response.ok) {
            return cachedResponse(request).then((cached) => cached ?? response);
          }
          const copy = response.clone();
          void caches.open(CACHE_NAME).then((cache) => cache.put(request, copy));
          return response;
        })
        .catch(() =>
          cachedResponse(request).then((cached) =>
            cached ?? cachedResponse(new Request(new URL('./index.html', self.location).href)),
          ),
        ),
    );
    return;
  }

  if (url.pathname === VERSION_MANIFEST_PATH) {
    event.respondWith(fetch(request, { cache: 'no-store' }));
    return;
  }

  event.respondWith(
    cachedResponse(request).then((cached) => {
      if (cached) return cached;

      return fetch(request).then((response) => {
        if (!response.ok) return response;
        const copy = response.clone();
        void caches.open(CACHE_NAME).then((cache) => cache.put(request, copy));
        return response;
      });
    }),
  );
});
